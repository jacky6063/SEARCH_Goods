**目標：** 實現客服人員接手/結束接手功能

#### 📝 API 規格

**端點：** `POST /api/repair/manual_mode`

**回應：**

**錯誤處理：**

```json
{
  "status": "error",

```

#### 🔧 實作要點（`backend/app.py`）

```python
class ManualModeRequest(BaseModel):
    session_id: str
    manual_mode: bool
    operator_id: Optional[str] = None
    operator_name: Optional[str] = None
    operator_avatar: Optional[str] = None  # 🆕

@app.post("/api/repair/manual_mode")
def toggle_repair_manual_mode(req: ManualModeRequest):
    """
    切換維修對話的人工模式
    
    流程：
    1. 檢查 session 是否存在
    2. 如果要開啟人工模式，檢查是否已被接手
    3. 更新 chat_sessions 表
    4. 記錄事件到 session_events
    5. 返回更新後的狀態
    """
    try:
        client = get_supabase_client(prefer_service_role=True)
        
        # 檢查現有狀態
        session_response = client.table('chat_sessions')\
            .select('session_id, manual_mode, operator_name')\
            .eq('session_id', req.session_id)\
            .eq('module_type', 'repair')\
            .single()\
            .execute()
        
        if not session_response.data:
            raise HTTPException(status_code=404, detail="Session 不存在")
        
        current_session = session_response.data
        
        # 防止重複接手
        if req.manual_mode and current_session.get('manual_mode'):
            raise HTTPException(
                status_code=409,
                detail=f"此對話已被 {current_session.get('operator_name')} 接手"
            )
        
        # 更新 session 狀態
        update_data = {
            "manual_mode": req.manual_mode,
            "operator_id": req.operator_id if req.manual_mode else None,
            "operator_name": req.operator_name if req.manual_mode else None,
            "operator_avatar": req.operator_avatar if req.manual_mode else None,
            "takeover_at": datetime.utcnow().isoformat() if req.manual_mode else None
        }
        
        client.table('chat_sessions')\
            .update(update_data)\
            .eq('session_id', req.session_id)\
            .execute()
        
        # 記錄事件
        REPAIR_LOGGING_BRIDGE.log_session_event(
            session_id=req.session_id,
            event_type="manual_mode_toggle",
            details={
                "manual_mode": req.manual_mode,
                "operator": req.operator_name,
                "timestamp": datetime.utcnow().isoformat()
            }
        )
        
        return {
            "status": "ok",
            "session_id": req.session_id,
            "manual_mode": req.manual_mode,
            "operator_name": req.operator_name,
            "operator_avatar": req.operator_avatar,
            "takeover_at": update_data.get("takeover_at")
        }
        
    except HTTPException:
        raise
    except Exception as e:
        logger.error(f"Toggle manual mode failed: {e}")
        raise HTTPException(status_code=500, detail=str(e))
```

```python
@app.post("/api/repair/session/{session_id}/reply")
def send_operator_reply(
    session_id: str,
    reply: str = Form(...),
    operator_id: str = Form(...),
    operator_name: Optional[str] = Form(None),
    operator_avatar: Optional[str] = Form(None),  // 🆕
    repair_items: Optional[str] = Form(None)
):
    """
    客服人員送出回覆
    
    檢查：
    1. Session 必須為人工模式
    2. 操作員 ID 需匹配（避免誤操作）
    """
    try:
        client = get_supabase_client(prefer_service_role=True)
        
        # 檢查 session 狀態
        session_check = client.table('chat_sessions')\
            .select('manual_mode, operator_id')\
            .eq('session_id', session_id)\
            .single()\
            .execute()
        
        if not session_check.data:
            raise HTTPException(status_code=404, detail="Session 不存在")
        
        if not session_check.data.get('manual_mode'):
            raise HTTPException(
                status_code=403,
                detail="此對話未處於人工模式，無法送出回覆"
            )
        
        # 記錄人工回覆
        payload = {
            "operator_id": operator_id,
            "operator_name": operator_name,
            "operator_avatar": operator_avatar,  # 🆕
            "manual_reply": True,
            "repair_items": json.loads(repair_items) if repair_items else []
        }
        
        message_record = client.table('chat_messages').insert({
            "session_id": session_id,
            "role": "operator",  # 🔥 關鍵角色標記
            "content": reply,
            "payload": payload,
            "source_module": "repair",
            "state": "processed",
            "created_at": datetime.utcnow().isoformat()
        }).execute()
        
        # 更新最後回覆時間
        client.table('chat_sessions')\
            .update({"last_operator_reply_at": datetime.utcnow().isoformat()})\
            .eq('session_id', session_id)\
            .execute()
        
        return {
            "status": "ok",
            "message_id": message_record.data[0]['message_id'],
            "session_id": session_id,
            "created_at": message_record.data[0]['created_at']
        }
        
    except HTTPException:
        raise
    except Exception as e:
        logger.error(f"Send operator reply failed: {e}")
        raise HTTPException(status_code=500, detail=str(e))
```

```python
def _check_repair_manual_mode(session_id: str) -> bool:
    """檢查 session 是否處於人工模式"""
```

```javascript
/**
 * 渲染單條訊息
 * @param {Object} message - 訊息物件
 * @param {string} message.role - 'user' | 'llm' | 'operator'
 * @param {string} message.content - 訊息內容
 * @param {Object} message.payload - 額外資訊
 */
function renderMessage(message) {
    const messageDiv = document.createElement('div');
    messageDiv.className = `message ${message.role}-message`;
    
    if (message.role === 'user') {
        // 使用者訊息（無頭像）
        messageDiv.innerHTML = `
            <div class="message-bubble">
                <div class="message-content">${escapeHtml(message.content)}</div>
                <div class="message-time">${formatTime(message.created_at)}</div>
            </div>
        `;
    } 
    else if (message.role === 'operator') {
        // 🔥 人工客服訊息
        const operatorName = message.payload?.operator_name || '客服人員';
        const operatorAvatar = message.payload?.operator_avatar || chatConfig.operatorAvatar;
        
        messageDiv.innerHTML = `
            <div class="avatar">
                <img src="${operatorAvatar}" 
                     alt="${operatorName}"
                     onerror="this.src='${chatConfig.operatorAvatar}'">
            </div>
            <div class="message-content-wrapper">
                <div class="role-label">客服 ${operatorName}</div>
                <div class="message-bubble">
                    <div class="message-content">${escapeHtml(message.content)}</div>
                    <div class="message-time">${formatTime(message.created_at)}</div>
                </div>
            </div>
        `;
    }
    else {
        // 🤖 AI 助手訊息
        messageDiv.innerHTML = `
            <div class="avatar">
                <img src="${chatConfig.aiAvatar}" 
                     alt="AI 助手"
                     onerror="this.src='${chatConfig.aiAvatar}'">
            </div>
            <div class="message-content-wrapper">
                <div class="role-label">AI 助手</div>
                <div class="message-bubble">
                    <div class="message-content">${escapeHtml(message.content)}</div>
                    <div class="message-time">${formatTime(message.created_at)}</div>
                </div>

}
```

### 任務 4.4：輪詢機制（檢測人工接手）

**目標：** 定期檢查新訊息，偵測人工接手狀態變化

#### 📝 實作要點

```javascript
let lastMessageTimestamp = null;
let isManualModeActive = false;
let pollingInterval = null;

/**
 * 開始輪詢新訊息
 */
function startMessagePolling(sessionId) {
    if (pollingInterval) {
        clearInterval(pollingInterval);
    }
    
    pollingInterval = setInterval(async () => {
        await pollNewMessages(sessionId);
    }, 3000);  // 每 3 秒檢查一次
}

/**
 * 輪詢新訊息
 */
async function pollNewMessages(sessionId) {
    try {
        const url = lastMessageTimestamp 
            ? `/api/repair/session/${sessionId}/messages?since=${lastMessageTimestamp}`
            : `/api/repair/session/${sessionId}/messages?limit=50`;
        
        const response = await fetch(url);
        const data = await response.json();
        
        for (const message of data.messages || []) {
            // 🔥 檢測到人工接手（從 llm → operator）
            if (!isManualModeActive && message.role === 'operator') {
                // 插入接手通知
                insertTakeoverNotification(
                    message.payload?.operator_name || '客服人員',
                    message.payload?.operator_avatar,
                    message.created_at
                );
                isManualModeActive = true;
            }
            
            // 渲染訊息
            const messageElement = renderMessage(message);
            document.getElementById('message-list').appendChild(messageElement);
            
            lastMessageTimestamp = message.created_at;
        }
        
        // 滾動到底部
        if (data.messages && data.messages.length > 0) {
            scrollToBottom();
        }
        
    } catch (error) {
        console.error('輪詢失敗', error);
    }
}

/**
 * 停止輪詢
 */
function stopMessagePolling() {
    if (pollingInterval) {
        clearInterval(pollingInterval);
        pollingInterval = null;
    }
}
```

### 任務 5.1：活躍對話列表

**目標：** 顯示所有待處理的對話

#### 📝 API 規格

**端點：** `GET /api/repair/active_sessions`

**查詢參數：**
- `limit` (int): 最多返回筆數，預設 50
- `status` (string): 狀態篩選，預設 "ongoing"

**回應：**
```json
{
  "sessions": [

      "manual_mode": false,
      "last_message": "水龍頭一直滴水...",
      "last_message_at": "2025-11-24T10:30:00Z",
      "message_count": 3,
      "operator_name": null
    },
    {
      "session_id": "def-456",
      "manual_mode": true,
      "last_message": "好的，請稍候...",
      "last_message_at": "2025-11-24T10:25:00Z",
      "message_count": 5,
      "operator_name": "王小明"
    }
  ],
  "total": 2
}
```

#### 🔧 後端實作

```python
@app.get("/api/repair/active_sessions")
def get_active_repair_sessions(
    limit: int = Query(50, ge=1, le=100),
    status: str = Query("ongoing")
):
    """
    取得活躍的維修對話列表
    
    用途：客服後台顯示待處理對話
    """
    try:
        client = get_supabase_client(prefer_service_role=True)
        
        # 查詢最近 30 分鐘的活躍 session
        cutoff_time = (datetime.utcnow() - timedelta(minutes=30)).isoformat()
        
        response = client.table('chat_sessions')\
            .select('session_id, manual_mode, operator_name, operator_avatar, started_at')\
            .eq('module_type', 'repair')\
            .eq('status', status)\
            .gte('started_at', cutoff_time)\
            .order('started_at', desc=True)\
            .limit(limit)\
            .execute()
        
        sessions = response.data or []
        
        # 查詢每個 session 的最後訊息
        enriched = []
        for session in sessions:
            sid = session['session_id']
            
            # 取得最後一條訊息
            msg_response = client.table('chat_messages')\
                .select('content, created_at')\
                .eq('session_id', sid)\
                .order('created_at', desc=True)\
                .limit(1)\
                .execute()
            
            last_msg = msg_response.data[0] if msg_response.data else {}
            
            # 計算訊息總數
            count_response = client.table('chat_messages')\
                .select('message_id', count='exact')\
                .eq('session_id', sid)\
                .execute()
            
            enriched.append({
                **session,
                "last_message": last_msg.get('content', ''),
                "last_message_at": last_msg.get('created_at', session['started_at']),
                "message_count": count_response.count or 0
            })
        
        return {
            "sessions": enriched,
            "total": len(enriched)
        }
        
    except Exception as e:
        logger.error(f"Get active sessions failed: {e}")
        return {"sessions": [], "total": 0}
```

**範例代碼（參考 `repair_chat_demo_operator.html`）：**

```javascript
// 接手對話
async function takeoverSession(sessionId) {
    try {
        // 取得客服資料（從 localStorage 或登入系統）
        const operatorProfile = getOperatorProfile();
        
        // 呼叫接手 API
        const response = await fetch('/api/repair/manual_mode', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify({
                session_id: sessionId,
                manual_mode: true,
                operator_id: operatorProfile.id,
                operator_name: operatorProfile.name,
                operator_avatar: operatorProfile.avatar
            })
        });
        
        if (!response.ok) {
            const error = await response.json();
            alert(`接手失敗: ${error.detail || error.message}`);
            return;
        }
        
        // 開啟對話視窗
        await openChatDialog(sessionId);
        
        // 開始輪詢
        startPolling(sessionId);
        
        showNotification('✅ 已接手對話', 'success');
        
    } catch (error) {
        console.error('Takeover failed:', error);
        alert('接手失敗：' + error.message);
    }
}

// 送出回覆
async function sendReply(sessionId) {
    const input = document.getElementById(`reply-input-${sessionId}`);
    const reply = input.value.trim();
    
    if (!reply) {
        alert('請輸入回覆內容');
        return;
    }
    
    try {
        const operatorProfile = getOperatorProfile();
        
        const formData = new FormData();
        formData.append('reply', reply);
        formData.append('operator_id', operatorProfile.id);
        formData.append('operator_name', operatorProfile.name);
        formData.append('operator_avatar', operatorProfile.avatar);
        
        const response = await fetch(`/api/repair/session/${sessionId}/reply`, {
            method: 'POST',
            body: formData
        });
        
        if (!response.ok) {
            const error = await response.json();
            throw new Error(error.detail || '送出失敗');
        }
        
        // 清空輸入框
        input.value = '';
        
        // 立即重新載入訊息
        await refreshMessages(sessionId);
        
        showNotification('✅ 回覆已送出', 'success');
        
    } catch (error) {
        console.error('Send reply failed:', error);
        alert('送出失敗：' + error.message);
    }
}
```
