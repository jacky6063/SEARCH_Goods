#### 1.1 前端請求階段

**使用者輸入場景**:
- 文字輸入: 「找女用背包」
- 語音輸入: 🎤 → Speech-to-Text → 「找女用背包」
- 分類點擊: UI 熱門分類 → `{L1: "包包配件", L2: "女用皮包", L3: "側背包"}`

```python
# backend/goods_search_service.py

def search_products_with_hierarchy(
    df: pd.DataFrame,
    query: str,
    hierarchy: Optional[Dict[str, str]] = None,
    topn: int = 10,
    min_score: float = 1.5,
) -> Tuple[List[Dict[str, Any]], List[str]]:
    """
    🚀 優化版: 利用分類索引加速搜尋
    
    流程：
    1. 使用分類索引篩選候選集 (O(1)) - 從 950 筆 → 50-100 筆
    2. 對候選集計分 (O(候選數 * 詞彙數))
    3. 執行其餘篩選和排序
    
    預期性能：搜尋時間從 200-500ms → 50-100ms
    """
    
    if hierarchy and any(hierarchy.get(k) for k in ["L1", "L2", "L3"]):
        # 🔍 使用分類索引 (Phase 1 優化)
        cat_index = get_category_index()
        candidate_indices = set(range(len(df)))
        
        # 逐層篩選候選集
        if hierarchy.get("L1"):
            l1_indices = cat_index.search_l1(hierarchy["L1"])
            if l1_indices:
                candidate_indices &= set(l1_indices)
        
        if hierarchy.get("L2"):
            l2_indices = cat_index.search_l2(hierarchy["L2"])
            if l2_indices:
                candidate_indices &= set(l2_indices)
        
        if hierarchy.get("L3"):
            l3_indices = cat_index.search_l3(hierarchy["L3"])
            if l3_indices:
                candidate_indices &= set(l3_indices)
        
        # 只對候選集操作 (大幅減少計算量)
        df_candidates = df.iloc[list(candidate_indices)].copy()
    else:
        df_candidates = df.copy()
    
    # 對候選集執行原始搜尋邏輯
    return search_products(df_candidates, query=query, topn=topn, min_score=min_score)
```

```python
def search_products(
    df: pd.DataFrame,
    query: str,
    topn: int = 10,
    min_score: float = 1.5,
    sort_price: bool = False,
    required_terms: Optional[List[str]] = None,
    category_terms: Optional[List[str]] = None,
    excluded_terms: Optional[List[str]] = None,
) -> Tuple[list[dict], List[str]]:
    """
    核心搜尋引擎 - 基於評分的商品排序
    
    評分演算法:
    • 商品名稱匹配: +3.0 分
    • REMARK 欄位匹配: +4.0 分
    • 描述/分類匹配: +1.5 分
    • 特價商品加成: +0.2 分
    • 最低門檻: 1.5 分
    
    優化點:
    • 🔧 Phase 2: 使用預計算文本快取 (__text_cache__)
    • 🔧 Phase 2: 批量篩選 (單個布林掩碼)
    • 🔧 Phase 2: 單鍵排序 (複合排序值)
    """
    
    # 提取搜尋詞彙
    terms = extract_terms(query)
    
    # 計算評分 (向量化操作)
    records = df.to_dict(orient="records")
    scores = [score_row(r, terms, query) for r in records]
    
    # 🔧 優化: 使用單個布林掩碼篩選
    sdf = df.copy()
    sdf["__score__"] = scores
    
    # 篩選條件
    mask = pd.Series([True] * len(sdf), index=sdf.index)
    
    # 必需詞彙篩選
    if required_terms:
        required_mask = sdf.apply(...)
        mask = mask & required_mask
    
    # 排除詞彙篩選
    if excluded_terms:
        excluded_mask = sdf.apply(...)
        mask = mask & ~excluded_mask
    
    # 一次性應用所有篩選
    filtered = sdf[mask]
    
    # 排序並返回
    sorted_df = filtered.sort_values("__score__", ascending=False)
    return sorted_df.head(topn).to_dict(orient="records"), terms
```

```python
def _filter_by_hierarchy(

    """

    • ⚡⚡ 超快速路徑: 熱門分類 UI L3 直接過濾 (from_hot_category=True)
    • ⚡ 快速路徑: L3 Only 直接過濾 (不指定 L1、L2)
    • 🔍 完整路徑: 逐層驗證 (L1 → L2 → L3)

    if from_hot_category and l3 and l1 and l2:

        filtered = [rec for rec in records if rec.get("CateName_L3") == l3]
        return filtered or records
    
    # 快速路徑：只指定 L3
    if l3 and not l1 and not l2:
        logger.info("    ⚡ 快速路徑（L3 Only）")
        filtered = [rec for rec in records if rec.get("CateName_L3") == l3]
        return filtered or records
    
    # 完整路徑：逐層驗證
    logger.info("    🔍 完整路徑（逐層驗證）")
    filtered = []
    for rec in records:
        ok = True
```

```python
# backend/llm_service.py

def llm_rerank_products(
    user_query: str,
    expanded_query: str,
    candidates: List[Dict[str, Any]],
    topn: int = 10,
    system_prompt: Optional[str] = None,
    use_search_config: bool = True,
) -> List[Dict[str, Any]]:
    """
    讓 LLM 基於語義相關性重新排序候選商品
    
    流程:
    1. 限制候選數量 (max_candidates = min(len, topn * 3))
    2. 構建商品摘要 JSON (id, name, category, price, description)
    3. 呼叫 OpenAI API 進行語義比對
    4. 解析 LLM 返回的排序結果
    5. 融合排序：結合 LLM 順序 + hierarchy_score
    
    範例輸出:
    {
      "matches": [
        {"id": "V55212D-1150", "score": 5, "reason": "完全符合女用側背包需求"},
        {"id": "V50201d-6003", "score": 4, "reason": "輕巧斜背包適合日常使用"}
      ]
    }
    """
    
    use_rerank = SEARCH_USE_RERANK if use_search_config else CHAT_USE_RERANK
    model = SEARCH_OPENAI_MODEL if use_search_config else CHAT_OPENAI_MODEL
    
    client = _get_client()
    if not use_rerank or not client or not candidates:
        return candidates[:topn]
    
    # 限制候選數量
    max_candidates = min(len(candidates), max(topn * 3, 15))
    subset = candidates[:max_candidates]
    
    # 構建商品摘要
    catalog = []
    for item in subset:
        catalog.append({
            "id": item.get("GoodIden") or "",
            "name": item.get("Name") or "",
            "category": item.get("CateName") or "",
            "price": item.get("Price") or "",
            "description": _truncate(item.get("DESCRIPTION") or "", 200)
        })
    
    # 呼叫 LLM
    payload = json.dumps(catalog, ensure_ascii=False)
    res = client.chat.completions.create(
        model=model,
        messages=[
            {"role": "system", "content": prompt},
            {"role": "user", "content": user_message}
        ],
        max_tokens=400,
        temperature=0.0
    )
    
    # 解析並返回重排結果
    parsed = json.loads(res.choices[0].message.content)
    matches = parsed.get("matches")
    
    # 根據 LLM 給出的順序重排
    reranked = []
    for entry in matches:
        pid = entry.get("id")
        chosen = id_to_item.get(pid)
        if chosen:
            reranked.append(chosen)
    
    return reranked[:topn]
```

### P0.3 TTL 快取層 (TTL Cache)

**目標**: 快取純分類查詢結果

```python
# backend/app.py

class TTLCache:
    """簡易 TTL 快取實現，使用 LRU 淘汰策略"""
    def __init__(self, max_size: int = 1000, ttl_seconds: int = 180):
        self.cache = OrderedDict()
        self.ttl = ttl_seconds
        self.max_size = max_size
    
    def get(self, key: str):
        """取得快取值，若過期則刪除並返回 None"""
        if key in self.cache:
            value, expire_time = self.cache[key]
            if time.time() < expire_time:
                self.cache.move_to_end(key)  # LRU
                return value
            else:
                del self.cache[key]
        return None
    
    def set(self, key: str, value):
        """設定快取值，若超過上限則移除最舊項目"""
        if len(self.cache) >= self.max_size:
            self.cache.popitem(last=False)
        self.cache[key] = (value, time.time() + self.ttl)

# 初始化快取
_category_cache = TTLCache(
    max_size=int(os.getenv("CATEGORY_CACHE_SIZE", "2000")),
    ttl_seconds=int(os.getenv("CATEGORY_CACHE_TTL", "180"))
)
```

**快取鍵範例**: `category:包包配件|女用皮包|側背包|1|10|False`

**命中率**: 預計 40-60% (熱門分類)

### P1.1 分類索引預建 (Category Index)

**目標**: O(1) 分類查詢取代 O(n) 掃描

```python
# backend/goods_search_service.py

class CategoryIndex:
    """
    分類層級索引，將 O(n) 掃描改為 O(1) 查詢
    
    範例：
    >>> index = CategoryIndex(df)
    >>> indices = index.search_l1("食品")  # O(1) - 立即返回
    >>> candidates = df.iloc[indices]
    >>> results = search_products(candidates, query)
    """
    
    def __init__(self, df: pd.DataFrame):
        """初始化索引 - 掃描所有行構建字典"""
        self.l1_index = self._build_level_index(df, ["CateName_L1", "大分類名稱"])
        self.l2_index = self._build_level_index(df, ["CateName_L2", "中分類名稱"])
        self.l3_index = self._build_level_index(df, ["CateName_L3", "小分類名稱"])
    
    def _build_level_index(self, df: pd.DataFrame, col_variants: List[str]) -> Dict[str, List[int]]:
        """
        構建索引字典
        
        返回: {分類名稱 → [行索引列表]}
        例如: {"食品": [0, 5, 10, ...], "衣物": [1, 3, 7, ...]}
        """
        # 找到存在的欄位
        col = next((c for c in col_variants if c in df.columns), None)
        if col is None:
            return {}
        
        # 構建索引
        index: Dict[str, List[int]] = {}
        for idx, row in df.iterrows():
            cat = str(row.get(col, "")).strip()
            if cat:
                index.setdefault(cat, []).append(int(idx))
        
        return index
    
    def search_l3(self, category_name: str) -> List[int]:
        """O(1) 查詢 L3 分類"""
        return self.l3_index.get(category_name.strip(), [])
```

**效能提升**: 70ms → 2ms (35 倍改進)
