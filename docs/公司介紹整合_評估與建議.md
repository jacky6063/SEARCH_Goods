## ⚠️ 需要改進的地方

### 1. **資料格式與設計文檔不一致**

**設計文檔建議**: JSON Lines 格式
```json
{
  "company_id": "ACME",
  "locale": "zh-TW",
  "company_name": "艾可米股份有限公司",
  "overview": "...",
  "contacts": {...},
  "faq": [...]
}
```

**現有格式**: CSV（平面結構）
- ❌ 缺少 `company_id` 和 `locale` 欄位
- ❌ 結構化資料（如聯絡方式）全部擠在描述文字中
- ❌ 難以程式化解析和查詢
- ❌ 不易擴展多公司/多語言

### 2. **欄位結構問題**

#### 問題 A: 關鍵字混雜在一起
```
現狀: "傳啟資訊，資訊系統整合，數位轉型顧問，智慧化企業方案..."
理想: ["傳啟資訊", "資訊系統整合", "數位轉型顧問", ...]
```

#### 問題 B: 聯絡資訊埋在描述中
```
現狀: "公司電話： 04-27062295 客服電話： 04-26062295..."
理想: {
  "service_phone": "+886-4-27062295",
  "customer_service_phone": "+886-4-26062295",
  ...
}
```

#### 問題 C: 缺少結構化的主題分類
```
現狀: 所有資訊都在「描述」欄位中
理想: 分為 overview, business_scope, organization, contacts, faq 等區塊
```

### 3. **查詢效率問題**

- ❌ CSV 格式需要全表掃描才能找到特定資訊
- ❌ 無法快速定位「客服電話」或「公司地址」
- ❌ 關鍵字搜尋需要字串比對，效率低

### 4. **可擴展性問題**

- ❌ 無法輕鬆新增其他公司資料（多品牌支援）
- ❌ 無法輕鬆新增多語言版本（繁中/英文）
- ❌ 新增欄位需要修改整個檔案結構

```python
"""
將現有的公司介紹 CSV 轉換為 JSON Lines 格式
"""
import csv
import json
import re
from pathlib import Path

def parse_contacts_from_description(description: str) -> dict:
    """從描述文字中提取聯絡資訊"""
    contacts = {}
    
    # 提取電話
    phone_match = re.search(r'公司電話：\s*(\d{2}-\d{8})', description)
    if phone_match:
        contacts['company_phone'] = f"+886-{phone_match.group(1)}"
    
    cs_phone_match = re.search(r'客服電話：\s*(\d{2}-\d{8})', description)
    if cs_phone_match:
        contacts['customer_service_phone'] = f"+886-{cs_phone_match.group(1)}"
    
    # 提取地址
    address_match = re.search(r'公司地址：\s*([^\n]+)', description)
    if address_match:
        contacts['address'] = address_match.group(1).strip()
    
    # 提取官網
    website_match = re.search(r'公司官網：\s*(https?://[^\s]+)', description)
    if website_match:
        contacts['website'] = website_match.group(1).strip()
    
    return contacts

def convert_csv_to_jsonl():
    csv_path = Path("../data/公司介紹.csv")
    output_path = Path("../data/company_profiles/company_profile_chuanchi.jsonl")
    
    # 確保輸出目錄存在
    output_path.parent.mkdir(parents=True, exist_ok=True)
    
    with open(csv_path, 'r', encoding='utf-8') as f:
        reader = csv.DictReader(f)
        
        for row in reader:
            category = row['資料類別']
            
            # 根據類別建立不同的資料結構
            if category == "公司基本資料":
                profile = {
                    "company_id": "chuanchi",
                    "locale": "zh-TW",
                    "company_name": "傳啟資訊股份有限公司",
                    "overview": row['描述'][:200] + "...",  # 摘要
                    "business_scope": row['關鍵字'].split('，'),
                    "media": {
                        "company_logo": row['代表圖片'],
                        "introduction_video": row['Youtube 影片介紹']
                    },
                    "keywords": row['關鍵字'].split('，')
                }
                
            elif category == "客服資訊":
                contacts = parse_contacts_from_description(row['描述'])
                profile = {
                    "company_id": "chuanchi",
                    "locale": "zh-TW",
                    "contacts": contacts
                }
            
            # 寫入 JSON Lines
            with open(output_path, 'a', encoding='utf-8') as out:
                json.dump(profile, out, ensure_ascii=False)
                out.write('\n')

if __name__ == "__main__":
    convert_csv_to_jsonl()
    print("✅ 轉換完成!")
```

"""
公司簡介查詢服務
"""
import json
from pathlib import Path
from typing import Dict, List, Optional, Any
import time

class CompanyProfileService:
    """公司簡介資料庫查詢服務"""
    
    def __init__(self):
        self.profiles: Dict[str, Dict[str, Any]] = {}
        self.index_by_keyword: Dict[str, List[str]] = {}
        self.loaded_at: Optional[float] = None
    
    def load_from_file(self, path: Path) -> None:
        """從 JSON Lines 檔案載入公司資料"""
        start = time.time()
        
        if not path.exists():
            print(f"[WARNING] Company profile file not found: {path}")
            return
        
        with open(path, 'r', encoding='utf-8') as f:
            for line in f:
                if not line.strip():
                    continue
                
                profile = json.loads(line)
                company_id = profile.get('company_id')
                locale = profile.get('locale', 'zh-TW')
                
                key = f"{company_id}:{locale}"
                
                # 合併資料（同一公司可能有多筆記錄）
                if key in self.profiles:
                    self.profiles[key].update(profile)
                else:
                    self.profiles[key] = profile
                
                # 建立關鍵字索引
                keywords = profile.get('keywords', [])
                for keyword in keywords:
                    if keyword not in self.index_by_keyword:
                        self.index_by_keyword[keyword] = []
                    if key not in self.index_by_keyword[keyword]:
                        self.index_by_keyword[keyword].append(key)
        
        self.loaded_at = time.time()
        elapsed = (self.loaded_at - start) * 1000
        
        print(f"[INFO] Company profiles loaded: {len(self.profiles)} profiles, "
              f"{len(self.index_by_keyword)} keywords, {elapsed:.1f}ms")
    
    def query(
        self, 
        company: str = "chuanchi", 
        topics: Optional[List[str]] = None,
        locale: str = "zh-TW"
    ) -> Optional[Dict[str, Any]]:
        """查詢公司資料"""
        key = f"{company}:{locale}"
        profile = self.profiles.get(key)
        
        if not profile:
            return None
        
        # 如果指定了主題，可以過濾相關內容
        if topics:
            # 這裡可以實作主題過濾邏輯
            pass
        
        return profile
    
    def get_contact_info(
        self, 
        company: str = "chuanchi",
        locale: str = "zh-TW"
    ) -> Optional[Dict[str, str]]:
        """快速查詢聯絡資訊"""
        profile = self.query(company, locale=locale)
        if profile:
            return profile.get('contacts', {})
        return None
    
    def search_by_keyword(self, keyword: str) -> List[Dict[str, Any]]:
        """根據關鍵字搜尋相關公司資訊"""
        keys = self.index_by_keyword.get(keyword, [])
        return [self.profiles[key] for key in keys if key in self.profiles]
    
    def get_faq(
        self,
        company: str = "chuanchi",
        category: Optional[str] = None,
        locale: str = "zh-TW"
    ) -> List[Dict[str, Any]]:
        """查詢 FAQ"""
        profile = self.query(company, locale=locale)
        if not profile:
            return []
        
        faqs = profile.get('faq', [])
        
        if category:
            faqs = [faq for faq in faqs if faq.get('category') == category]
        
        return faqs

# 全域單例
_company_profile_service: Optional[CompanyProfileService] = None

def get_company_profile_service() -> CompanyProfileService:
    """取得公司簡介服務單例"""
    global _company_profile_service
    if _company_profile_service is None:
        _company_profile_service = CompanyProfileService()
    return _company_profile_service

    query: str,
    profile: Dict[str, Any],
    topics: List[str]
) -> Dict[str, Any]:
    """建立公司資訊回覆"""
    
    # 判斷查詢類型
    query_lower = query.lower()
    
    if any(word in query_lower for word in ["客服", "電話", "聯絡"]):
        # 客服資訊
