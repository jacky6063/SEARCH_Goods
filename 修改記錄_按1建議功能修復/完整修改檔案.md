def get_df() -> pd.DataFrame:
    """取得商品資料，支援快取機制"""
    global _df_cache, _cache_timestamp
    
    if not os.path.exists(DATA_PATH):
        raise FileNotFoundError(f"Data file not found: {DATA_PATH}")
    
    file_mtime = os.path.getmtime(DATA_PATH)
    if _df_cache is None or file_mtime > _cache_timestamp:
        try:
            _df_cache = pd.read_csv(DATA_PATH, encoding="utf-8")
            _cache_timestamp = file_mtime
            logger.info(f"Loaded CSV with {len(_df_cache)} rows from {DATA_PATH}")
        except Exception as e:
            logger.error(f"Failed to load CSV: {e}")
            if _df_cache is None:
                raise
    
    return _df_cache

def _cleanup_session_cache(now: float):
    """清理過期的 session cache"""
    expired_sessions = []
    for session_id, cache_data in SUGGEST_CACHE.items():
        if now - cache_data.get("ts", 0) > SESSION_TIMEOUT:
            expired_sessions.append(session_id)
    
    for session_id in expired_sessions:
        del SUGGEST_CACHE[session_id]
    
    expired_align = []
    for session_id, align_data in SESSION_ALIGN_CACHE.items():
        if now - align_data.get("ts", 0) > SESSION_TIMEOUT:
            expired_align.append(session_id)
    
    for session_id in expired_align:
        del SESSION_ALIGN_CACHE[session_id]

def _parse_price(row: Dict[str, Any]) -> float:
    """解析商品價格"""
    price_keys = ["SpecialOffer", "特價", "pric_special", "Price", "價格", "pric"]
    for key in price_keys:
        if key in row:
            value = str(row.get(key) or "").replace(",", "").strip()
            if value:
                try:
                    return float(value)
                except Exception:
                    continue
    return 0.0

class SuggestReq(BaseModel):
    session_id: Optional[str] = None
    type: int = 1

def _extract_ids_from_items(items: List[Dict[str, Any]]) -> List[str]:
    """從商品列表中提取 ID"""
    ids: List[str] = []
    for item in items or []:
        if not isinstance(item, dict):
            continue
        val = (
            item.get("商品編號")
            or item.get("GoodIden")
            or item.get("id")
            or item.get("GoodId")
            or item.get("good_id")
        )
        if not val:
            continue
        ident = str(val).strip()
        if ident:
            ids.append(ident)
    return ids

def _filter_on_sale(items: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
    """篩選特價商品"""
    filtered: List[Dict[str, Any]] = []
    for item in items or []:
        if not isinstance(item, dict):
            continue
        special = (
            item.get("商品特價")
            or item.get("SpecialOffer")
            or item.get("special")
            or item.get("特價")
        )
        if special in (None, "", 0, "0"):
            continue
        try:
            numeric = float(str(special).replace(",", "").strip())
            if numeric <= 0:
                continue
        except Exception:
            pass
        filtered.append(item)
    return filtered

def _build_suggestion(session_id: str, suggestion_type: int, df: pd.DataFrame):
    """建立建議商品列表"""
    cache = SUGGEST_CACHE.get(session_id) or {}
    align_ids = cache.get("align_ids") or []
    align_rows = cache.get("align_rows") or []
    query_terms = cache.get("query_terms") or []
    structured_items = cache.get("structured_items") or []
    structured_summary = cache.get("structured_summary") or ""

    if suggestion_type == 1:
        # 原始推薦：優先使用 structured_items
        if structured_items:
            ids = _extract_ids_from_items(structured_items) or align_ids
            rows = structured_items
            return ids, rows, structured_summary or f"為您準備 {len(rows)} 項商品建議"
        ids = suggest_original_ids(align_ids)
    elif suggestion_type == 2:
        # 特價推薦：從 structured_items 中篩選特價商品
        sale_items = _filter_on_sale(structured_items)
        if sale_items:
            ids = _extract_ids_from_items(sale_items)
            rows = sale_items
            return ids, rows, f"為您準備 {len(rows)} 項特價商品推薦"
        ids = suggest_on_sale_related(df, query_terms)
    else:
        # 互補推薦
        ids = suggest_complementary(df, align_rows)

    # 從 DataFrame 取得商品資料
    rows = get_items_by_ids(df, ids)
    
    # 如果沒有找到商品且有 structured_items，使用 structured_items
    if not rows and structured_items:
        rows = structured_items
        if not ids:
            ids = _extract_ids_from_items(structured_items)
    
    summary = structured_summary or f"為您準備 {len(rows)} 項商品建議"
    return ids, rows, summary

@app.post("/api/chat", response_model=ChatResp)
def chat_endpoint(req: ChatReq):
    try:
        logger.info(f"Chat request: {req.message[:100]}...")
        
        # 取得商品資料
        df = get_df()
        
        # 呼叫聊天處理器
        result = chat_handler(
            message=req.message,
            session_id=req.session_id,
            df=df,
            history=req.history or []
        )
        
        # 🔧 修正：統一處理回傳格式 - 支援 ChatResponse 對象
        if hasattr(result, 'reply'):  # ChatResponse 對象
            # 確保總是有 session_id，如果沒有就生成一個  
            session_id = getattr(result, 'chat_session_id', None) or req.session_id
            if not session_id:
                import uuid
                session_id = str(uuid.uuid4())[:8]
            
            # 🔧 修正：同步到 SUGGEST_CACHE - 支援新的 structured_products 格式
            # 注意：chat_handler 返回的是 ChatResponse 對象，使用屬性訪問
            structured_payload = getattr(result, 'structured_payload', None)
            structured_products = structured_payload.get("items", []) if structured_payload else []
            
            try:
                suggestion_ids = getattr(result, 'suggestion_ids', None) or []
                
                # 🔧 修正：放寬建立 cache 的條件
                if suggestion_ids or structured_products:
                    df = get_df()
                    align_rows = get_items_by_ids(df, suggestion_ids) if suggestion_ids else []
                    
                    SUGGEST_CACHE[session_id] = {
                        "align_ids": suggestion_ids,
                        "align_rows": align_rows,
                        "query_terms": [req.message],
                        "structured_items": structured_products,  # 🆕 存儲結構化商品資料
                        "structured_summary": structured_payload.get("summary", f"為您找到了 {len(structured_products)} 款相關商品") if structured_payload else "",
                        "ts": time.time(),
                    }
                    logger.info(f"🔧 Updated SUGGEST_CACHE for session {session_id}: {len(structured_products)} structured_items")
            except Exception as e:
                logger.warning(f"Failed to sync SUGGEST_CACHE: {e}")
                    
            return ChatResp(
                reply=getattr(result, 'reply', ''),
                suggestion_ids=suggestion_ids,
                session_id=session_id,
                structured_products=structured_products  # 🆕 傳遞結構化商品資料
            )
        elif isinstance(result, dict):
            # 處理字典格式的回傳（向後兼容）
            session_id = result.get("session_id") or req.session_id
            if not session_id and result.get("suggestion_ids"):
                import uuid
                session_id = str(uuid.uuid4())[:8]
                # 同步到 SUGGEST_CACHE
                try:
                    suggestion_ids = result.get("suggestion_ids", [])
                    if suggestion_ids:
                        df = get_df()
                        rows = get_items_by_ids(df, suggestion_ids)
                        SUGGEST_CACHE[session_id] = {
                            "align_ids": suggestion_ids,
                            "align_rows": rows,
                            "query_terms": [req.message],
                            "ts": time.time(),
                        }
                except Exception as e:
                    logger.warning(f"Failed to sync SUGGEST_CACHE: {e}")
                        
            return ChatResp(
                reply=result.get("reply", ""),
                suggestion_ids=result.get("suggestion_ids", []),
                session_id=session_id,
                structured_products=result.get("structured_products", [])
            )
        else:
            # 處理純字串回傳
            return ChatResp(reply=str(result), session_id=req.session_id)
            
    except Exception as e:
        logger.error(f"Chat error: {e}", exc_info=True)
        return ChatResp(reply="抱歉，處理您的請求時發生錯誤，請稍後再試。", session_id=req.session_id)

@app.post("/api/suggest")
def suggest_endpoint(req: SuggestReq):
    session_id = str(req.session_id or "default")
    suggestion_type = int(req.type or 1)
    now = int(time.time())
    
    # 清理過期的 cache
    _cleanup_session_cache(now)
    
    df = get_df()
    ids, rows, summary = _build_suggestion(session_id, suggestion_type, df)
    
    if not rows:
        return {"mode": "chat", "items": [], "products": []}
    
    # 🔧 修正：為向後兼容，同時提供 items (中文格式) 和 products (英文格式)
    items = format_for_chat(rows)
    products = []
    
    for row in rows:
        if isinstance(row, dict):
            # 處理中文格式商品 (來自 structured_items)
            if "商品名稱" in row:
                products.append({
                    "id": row.get("商品編號", ""),
                    "name": row.get("商品名稱", ""),
                    "price": str(row.get("商品價格", "")),
                    "special_price": str(row.get("商品特價", "")),
                    "description": row.get("商品描述", ""),
                    "image_url": row.get("商品圖片網址", ""),
                    "shop_url": row.get("商品購物網址", ""),
                })
            # 處理英文格式商品 (來自 DataFrame)
            else:
                products.append({
                    "id": row.get("GoodIden", ""),
                    "name": row.get("Name", ""),
                    "price": str(row.get("Price", "")),
                    "special_price": str(row.get("SpecialOffer", "")),
                    "description": row.get("DESCRIPTION", ""),
                    "image_url": row.get("Goodspic_Link1", ""),
                    "shop_url": row.get("Goods_Link1", ""),
                })
    
    logger.info(f"🔧 Suggest type {suggestion_type} for session {session_id}: {len(rows)} rows, {len(products)} products")
    
    return {
        "mode": "render",
        "items": items,  # 保持向後兼容
        "products": products,  # 🆕 新增英文格式
        "ids": ids,
        "message": summary or f"為您準備 {len(rows)} 項商品建議",
        "assistant_reply": summary or f"為您準備 {len(rows)} 項商品建議",
    }

# 管理 API
def _verify_admin_token(token: str):
    if ALLOW_DEV_ADMIN:
        return True
    return token == ADMIN_TOKEN

@app.post("/api/admin/upload-csv")
def upload_csv(file: UploadFile = File(...), admin_token: str = Form(...)):
    if not _verify_admin_token(admin_token):
        raise HTTPException(status_code=401, detail="Invalid admin token")
    
    if not file.filename.endswith('.csv'):
        raise HTTPException(status_code=400, detail="Only CSV files are allowed")
    
    try:
        # 原子性檔案替換
        temp_path = f"{DATA_PATH}.temp"
        with open(temp_path, "wb") as buffer:
            content = file.file.read()
            buffer.write(content)
        
        # 驗證 CSV 格式
        pd.read_csv(temp_path, encoding="utf-8", nrows=1)
        
        # 原子性移動
        os.replace(temp_path, DATA_PATH)
        
        # 清除快取
        global _df_cache, _cache_timestamp
        _df_cache = None
        _cache_timestamp = 0.0
        
        return {"success": True, "message": f"Successfully uploaded {file.filename}"}
    except Exception as e:
        # 清理臨時檔案
        if os.path.exists(temp_path):
            os.remove(temp_path)
        logger.error(f"CSV upload error: {e}")
        raise HTTPException(status_code=500, detail=f"Upload failed: {str(e)}")

@app.post("/api/admin/clear-cache")
def clear_cache(admin_token: str = Form(...)):

    _df_cache = None

    SUGGEST_CACHE.clear()
    SESSION_ALIGN_CACHE.clear()

- 新增 `structured_items` 和 `structured_summary` 存儲

### 3. 建議 API 格式轉換支援 (第 310-350 行)
