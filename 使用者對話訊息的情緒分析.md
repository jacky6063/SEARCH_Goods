# 使用者對話訊息的情緒分析

## 📋 功能概述

在住宅維修客服系統中，自動分析使用者訊息的情緒狀態，標示出：
- 😰 **不安程度**（焦慮、擔心、害怕）
- ⚡ **急迫感**（緊急、需立即處理）
- 😠 **生氣指數**（憤怒、不滿、抱怨）

當情緒指數達到設定標準時，將分析結果記錄到資料庫，並可觸發自動通知或優先處理機制。

---

## 🎯 核心需求

### 1. 情緒分析結果儲存
- ✅ 將 LLM 判讀的情緒指數記錄到使用者回覆訊息的資料欄位
- ✅ 只記錄達到標準的訊息（節省儲存空間）
- ✅ 支援歷史資料回溯查詢

### 2. 外部提示詞設定
- ✅ LLM 評分標準可透過外部配置文件設定
- ✅ 支援動態調整閾值（不需重啟服務）
- ✅ 可針對不同維修類別設定不同標準

```sql
CREATE TABLE emotion_analysis (
    analysis_id SERIAL PRIMARY KEY,
    message_id INTEGER REFERENCES chat_messages(message_id),
    session_id VARCHAR(255) NOT NULL,
    anxiety_level INTEGER CHECK (anxiety_level BETWEEN 0 AND 10),
    urgency_level INTEGER CHECK (urgency_level BETWEEN 0 AND 10),
    anger_level INTEGER CHECK (anger_level BETWEEN 0 AND 10),
    keywords TEXT[],
    reasoning TEXT,
    analyzed_at TIMESTAMP DEFAULT NOW(),
    config_version VARCHAR(50),
    INDEX idx_urgency (urgency_level DESC),
    INDEX idx_session (session_id)
);
```

### 1. 配置管理模組 (`backend/config_emotion.py`)

```python
import json
import os
from datetime import datetime
from typing import Dict, Optional
from pathlib import Path

class EmotionAnalysisConfig:
    """情緒分析配置管理"""
    
    def __init__(self, config_path: str = "backend/config/emotion_analysis_config.json"):
        self.config_path = config_path
        self.config = self.load_config()
        self.last_loaded = datetime.now()
    
    def load_config(self) -> dict:
        """載入配置文件"""
        if not os.path.exists(self.config_path):
            return self._get_default_config()
        
        with open(self.config_path, 'r', encoding='utf-8') as f:
            return json.load(f)
    
    def reload_if_needed(self):
        """檢查文件是否更新，自動重新載入"""
        if os.path.exists(self.config_path):
            file_mtime = datetime.fromtimestamp(os.path.getmtime(self.config_path))
            if file_mtime > self.last_loaded:
                self.config = self.load_config()
                self.last_loaded = datetime.now()
                print(f"[EmotionConfig] 配置已重新載入: {file_mtime}")
    
    def get_threshold(self, emotion_type: str) -> int:
        """取得記錄閾值"""
        return self.config.get('thresholds', {}).get(emotion_type, 7)
    
    def get_prompt_template(self) -> str:
        """取得 LLM Prompt 模板"""
        return self.config.get('llm_prompt_template', '')
    
    def get_scoring_criteria(self) -> str:
        """取得評分標準（格式化為文字）"""
        criteria = self.config.get('scoring_criteria', {})
        formatted = []
        for emotion, levels in criteria.items():
            formatted.append(f"\n【{emotion.upper()} 評分標準】")
            for level, desc in levels.items():
                formatted.append(f"- {level}: {desc}")
        return '\n'.join(formatted)
    
    def should_record(self, anxiety: int, urgency: int, anger: int) -> bool:
        """判斷是否達到記錄標準"""
        thresholds = self.config.get('thresholds', {})
        return (
            anxiety >= thresholds.get('anxiety', 7) or
            urgency >= thresholds.get('urgency', 8) or
            anger >= thresholds.get('anger', 6)
        )
    
    def get_auto_action(self, emotion_type: str, level: int) -> Optional[dict]:
        """取得自動化動作配置"""
        actions = self.config.get('auto_actions', {})
        key = f"{emotion_type}_{level}_10" if level >= 9 else None
        return actions.get(key)
    
    def _get_default_config(self) -> dict:
        """預設配置"""
        return {
            "version": "1.0.0",
            "thresholds": {"anxiety": 7, "urgency": 8, "anger": 6}
        }

# 全域配置實例
emotion_config = EmotionAnalysisConfig()
```

### 2. 情緒分析服務 (`backend/emotion_service.py`)

```python
import json
from typing import Dict, Optional
from datetime import datetime
from openai import OpenAI
from config_emotion import emotion_config

client = OpenAI(api_key=os.getenv('OPENAI_API_KEY'))

def analyze_user_emotion(message: str, session_id: str) -> Optional[Dict]:
    """
    分析使用者訊息情緒
    
    Args:
        message: 使用者訊息內容
        session_id: 會話 ID
    
    Returns:
        情緒分析結果（達到標準才回傳，否則 None）
    """
    try:
        # 重新載入配置（支援熱更新）
        emotion_config.reload_if_needed()
        
        # 組合 Prompt
        prompt = emotion_config.get_prompt_template().format(
            message=message,
            criteria=emotion_config.get_scoring_criteria()
        )
        
        # 呼叫 LLM
        response = client.chat.completions.create(
            model=os.getenv('OPENAI_MODEL', 'gpt-4o-mini'),
            messages=[
                {"role": "system", "content": "你是一個專業的情緒分析助手，專門分析客服對話中的情緒狀態。"},
                {"role": "user", "content": prompt}
            ],
            temperature=0.3,
            response_format={"type": "json_object"}
        )
        
        result = json.loads(response.choices[0].message.content)
        
        # 檢查是否達到記錄標準
        anxiety = result.get('anxiety_level', 0)
        urgency = result.get('urgency_level', 0)
        anger = result.get('anger_level', 0)
        
        if not emotion_config.should_record(anxiety, urgency, anger):
            return None  # 未達標準，不記錄
        
        # 補充元數據
        result['analyzed_at'] = datetime.now().isoformat()
        result['config_version'] = emotion_config.config.get('version', '1.0.0')
        result['trigger_threshold'] = {
            'anxiety': emotion_config.get_threshold('anxiety'),
            'urgency': emotion_config.get_threshold('urgency'),
            'anger': emotion_config.get_threshold('anger')
        }
        
        # 檢查是否需要自動化動作
        if urgency >= 9:
            action = emotion_config.get_auto_action('urgency', urgency)
            if action:
                result['auto_action'] = action
                # 這裡可以觸發通知機制
                trigger_notification(session_id, action)
        
        return result
        
    except Exception as e:
        print(f"[EmotionAnalysis] 分析失敗: {str(e)}")
        return None


def trigger_notification(session_id: str, action: dict):
    """觸發自動化通知（預留接口）"""
    print(f"[AUTO_ACTION] Session {session_id}: {action['message']}")
```

### 3. 整合到聊天 API (`backend/app.py`)

```python
from emotion_service import analyze_user_emotion

@app.post("/api/repair/chat")
async def repair_chat(request: RepairChatRequest):
    """住宅維修聊天（含情緒分析）"""
    
    # 1. 儲存使用者訊息
    user_msg_id = await db.save_message(
        session_id=request.session_id,
        role='user',
        content=request.message
    )
    
    # 2. 非同步進行情緒分析
    emotion_result = analyze_user_emotion(
        message=request.message,
        session_id=request.session_id
    )
    
    # 3. 如果達到標準，更新訊息記錄
    if emotion_result:
        await db.update_message_emotion(user_msg_id, emotion_result)
        
        # 記錄到日誌
        print(f"[EMOTION] Message #{user_msg_id} | "
              f"Anxiety:{emotion_result['anxiety_level']} "
              f"Urgency:{emotion_result['urgency_level']} "
              f"Anger:{emotion_result['anger_level']}")
    
    # 4. 檢查是否為人工接手模式
    manual_mode = get_manual_mode(request.session_id)
    if manual_mode:
        return {"reply": "[此對話已由真人客服接手]", "mode": "manual"}
    
    # 5. LLM 回覆（可根據情緒調整語氣）
    reply = await generate_llm_reply(
        session_id=request.session_id,
        message=request.message,
        emotion_hint=emotion_result  # 傳遞情緒資訊
    )
    
    return {"reply": reply, "mode": "ai", "emotion": emotion_result}
```

```python
async def update_message_emotion(message_id: int, emotion_data: dict):
    """更新訊息的情緒分析資料"""
    query = """
        UPDATE chat_messages 
        SET emotion_data = $1
        WHERE message_id = $2
    """
    await db.execute(query, json.dumps(emotion_data), message_id)


async def get_high_emotion_messages(date: str, emotion_type: str, min_level: int):
    """查詢高情緒訊息"""
    query = """
        SELECT message_id, session_id, content, emotion_data, created_at
        FROM chat_messages
        WHERE DATE(created_at) = $1
          AND role = 'user'
          AND emotion_data IS NOT NULL
          AND (emotion_data->>$2)::int >= $3
        ORDER BY (emotion_data->>$2)::int DESC
    """
    return await db.fetch(query, date, f"{emotion_type}_level", min_level)
```

```javascript
function applyEmotionFilters() {
    const showUrgent = document.getElementById('filter-high-urgency').checked;
    const showAnxious = document.getElementById('filter-high-anxiety').checked;
    const showAngry = document.getElementById('filter-angry').checked;
    
    document.querySelectorAll('.message-card.user').forEach(card => {
        const emotionData = card.dataset.emotion 
            ? JSON.parse(card.dataset.emotion) 
            : null;
        
        if (!emotionData) {
            card.style.display = (showUrgent || showAnxious || showAngry) ? 'none' : 'block';
            return;
        }
        
        const match = (
            (!showUrgent || emotionData.urgency_level >= 8) &&
            (!showAnxious || emotionData.anxiety_level >= 7) &&
            (!showAngry || emotionData.anger_level >= 6)
        );
        
        card.style.display = match ? 'block' : 'none';
    });
```

## ⚠️ 注意事項與風險

### 隱私與合規
- ✅ 情緒分析結果僅供內部使用，不回傳給使用者
- ✅ 遵守 GDPR/個資法，不外洩訊息內容
- ⚠️ 建議在服務條款中說明「使用 AI 分析服務品質」

### 技術風險
- ⚠️ LLM 成本：建議設定每日分析上限（例如 500 則）
- ⚠️ 分析延遲：非同步處理避免影響對話流程
- ⚠️ 誤判風險：定期人工校正（建議每月標註 50 筆）

### 文化差異
- ⚠️ 中文表達習慣：「...」可能表示無奈或思考，非生氣
- ⚠️ 地區用語：台灣、香港、大陸用詞差異需調整關鍵字庫
- ✅ 建議：累積 200 筆資料後，針對台灣維修情境調整 Prompt
